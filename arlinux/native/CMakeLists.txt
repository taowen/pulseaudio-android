# Probe the target NDK, never the Windows host headers or functions.
foreach(header arpa/inet.h byteswap.h dlfcn.h dirent.h endian.h fcntl.h grp.h inttypes.h langinfo.h
    linux/sockios.h locale.h memory.h netdb.h netinet/in.h netinet/in_systm.h netinet/ip.h
    netinet/tcp.h poll.h pthread.h pwd.h regex.h sched.h signal.h stdint.h stdlib.h
    strings.h string.h sys/eventfd.h sys/ioctl.h sys/mman.h sys/prctl.h sys/resource.h
    sys/select.h sys/socket.h sys/stat.h sys/syscall.h sys/time.h sys/types.h sys/uio.h
    sys/un.h sys/utsname.h sys/wait.h syslog.h unistd.h)
  string(MAKE_C_IDENTIFIER "HAVE_${header}" id)
  string(TOUPPER "${id}" id)
  check_include_file("${header}" ${id})
  if(${id})
    feature(${id} 1)
  endif()
endforeach()
foreach(func accept4 calloc ceil clock_gettime ctime_r dladdr dlerror fchmod fchown floor fmod fork
    free fstat fstat64 fsync ftruncate getaddrinfo getgrgid_r getgrnam_r getpagesize getpwnam_r
    getpwuid_r gettimeofday getuid gmtime gmtime_r localtime localtime_r lrint lrintf lround lseek
    lseek64 lstat malloc mlock mmap nanosleep open open64 pipe pipe2 posix_fadvise posix_madvise
    posix_memalign ppoll pthread_getname_np pthread_setaffinity_np pthread_setname_np read readlink
    realloc setegid seteuid setlocale setpgid setregid setresgid setresuid setreuid setsid sigaction
    snprintf strerror_r strlcat strlcpy strtod_l strtof symlink sysconf uname vsnprintf waitpid write)
  string(TOUPPER "HAVE_${func}" id)
  check_function_exists(${func} ${id})
  if(${id})
    feature(${id} 1)
  endif()
endforeach()

function(pulse_target name)
  target_include_directories(${name} PRIVATE "${GEN}" "${PULSE}/src" "${GEN}/sndfile" "${LTDL}")

function(pulse_module name source)
  add_library(${name} SHARED "${source}")
  pulse_target(${name})
  string(REPLACE "-" "_" module_id "${name}")
  target_compile_definitions(${name} PRIVATE __INCLUDED_FROM_PULSE_AUDIO PA_MODULE_NAME=${module_id})
  target_link_libraries(${name} PRIVATE pulsecore pulse pulsecommon)
  set_target_properties(${name} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${OUTPUT}/assets/pulseaudio")
endfunction()
